cmake_minimum_required(VERSION 3.28)
project(obs-scenepad VERSION 3.1.0 LANGUAGES CXX)
if(NOT WIN32 AND NOT APPLE)
  message(FATAL_ERROR "ScenePad requires Windows or macOS.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ScenePad requires a 64-bit build.")
endif()
if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
  message(FATAL_ERROR "ScenePad macOS builds require arm64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release)
find_package(libobs REQUIRED CONFIG)
find_package(obs-frontend-api REQUIRED CONFIG)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network)
if(WIN32)
  set(ZLIB_USE_STATIC_LIBS ON)
endif()
find_package(ZLIB REQUIRED)
set(CMAKE_AUTORCC ON)

# Share the implementation between the plugin and its tests.
add_library(scenepad-core STATIC
  src/dock.cpp src/editor.cpp src/notes.cpp src/backup.cpp src/docx.cpp src/icons.cpp
  src/remote.cpp src/web-document.cpp src/vendor/qrcodegen.cpp src/remote.qrc)
target_include_directories(scenepad-core PUBLIC src)
target_link_libraries(scenepad-core PUBLIC OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network ZLIB::ZLIB)
target_compile_definitions(scenepad-core PUBLIC SCENEPAD_VERSION="${PROJECT_VERSION}")
set_target_properties(scenepad-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  target_link_libraries(scenepad-core PUBLIC ole32 uuid)
  target_compile_definitions(scenepad-core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(scenepad-core PUBLIC /W4 /utf-8 /permissive-)
else()
  find_library(SYSTEM_ARCHIVE archive REQUIRED)
  find_path(LIBARCHIVE_INCLUDE_DIR archive.h REQUIRED)
  target_include_directories(scenepad-core PRIVATE "${LIBARCHIVE_INCLUDE_DIR}")
  target_link_libraries(scenepad-core PUBLIC "${SYSTEM_ARCHIVE}")
  target_compile_options(scenepad-core PUBLIC -Wall -Wextra -Wpedantic)
endif()

add_library(obs-scenepad MODULE src/plugin.cpp)
target_link_libraries(obs-scenepad PRIVATE scenepad-core)
set_target_properties(obs-scenepad PROPERTIES PREFIX "" OUTPUT_NAME "obs-scenepad")
if(WIN32)
  configure_file(src/version.rc.in version.rc @ONLY)
  target_sources(obs-scenepad PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
  install(TARGETS obs-scenepad RUNTIME DESTINATION obs-scenepad/bin/64bit LIBRARY DESTINATION obs-scenepad/bin/64bit)
  set(resource_path obs-scenepad/data)
else()
  configure_file(src/Info.plist.in Info.plist @ONLY)
  set_target_properties(obs-scenepad PROPERTIES
    BUNDLE TRUE BUNDLE_EXTENSION plugin
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER net.haliluya.scenepad
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/Info.plist"
    BUILD_WITH_INSTALL_RPATH FALSE
    INSTALL_RPATH "@executable_path/../Frameworks")
  install(TARGETS obs-scenepad LIBRARY DESTINATION .)
  set(resource_path obs-scenepad.plugin/Contents/Resources)
endif()
install(DIRECTORY data/ DESTINATION "${resource_path}")
install(FILES LICENSE README.md DESTINATION "${resource_path}")
install(FILES LICENSES/qrcodegen.txt DESTINATION "${resource_path}/licenses")

include(CTest)
if(BUILD_TESTING)
  if(APPLE)
    set(test_platform cocoa)
  else()
    set(test_platform windows)
  endif()
  add_executable(scenepad-tests tests/test-main.cpp)
  target_link_libraries(scenepad-tests PRIVATE scenepad-core)
  target_compile_definitions(scenepad-tests PRIVATE SCENEPAD_LOCALE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/locale")
  foreach(locale IN ITEMS en-US zh-CN zh-TW)
    if(locale STREQUAL "en-US")
      set(scroll_label "Scroll")
    elseif(locale STREQUAL "zh-CN")
      set(scroll_label "滚动")
    else()
      set(scroll_label "捲動")
    endif()
    add_test(NAME scenepad-${locale} COMMAND scenepad-tests
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.docx"
      "${CMAKE_CURRENT_SOURCE_DIR}/data/locale/${locale}.ini" "${scroll_label}")
    set_tests_properties(scenepad-${locale} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${test_platform}" TIMEOUT 60)
  endforeach()
  add_executable(scenepad-stress tests/stress.cpp)
  target_link_libraries(scenepad-stress PRIVATE scenepad-core)
  if(WIN32)
    target_link_libraries(scenepad-stress PRIVATE psapi)
  endif()
  add_executable(scenepad-remote-tests tests/remote.cpp)
  target_link_libraries(scenepad-remote-tests PRIVATE scenepad-core)
  add_test(NAME scenepad-remote COMMAND scenepad-remote-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.docx")
  set_tests_properties(scenepad-remote PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${test_platform}" TIMEOUT 60)
  add_executable(scenepad-notes-tests tests/notes.cpp)
  target_link_libraries(scenepad-notes-tests PRIVATE scenepad-core)
  add_test(NAME scenepad-notes COMMAND scenepad-notes-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.docx")
  set_tests_properties(scenepad-notes PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${test_platform}" TIMEOUT 180)
  add_executable(scenepad-limits-tests tests/limits.cpp)
  target_link_libraries(scenepad-limits-tests PRIVATE scenepad-core)
  add_test(NAME scenepad-limits COMMAND scenepad-limits-tests)
  set_tests_properties(scenepad-limits PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${test_platform}" TIMEOUT 60)
  add_executable(scenepad-autoclear-tests tests/auto-clear.cpp)
  target_link_libraries(scenepad-autoclear-tests PRIVATE scenepad-core)
  add_test(NAME scenepad-autoclear COMMAND scenepad-autoclear-tests)
  set_tests_properties(scenepad-autoclear PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${test_platform}" TIMEOUT 60)
endif()
